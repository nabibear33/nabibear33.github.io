# 4.1 Pointers

## 포인터를 잘 사용하는 방법
- 객체를 가리키지 않는 포인터는 NULL로 설정한다.
- type cast를 명확하게 한다.
    
    pi = malloc(sizeof(int));

    pf = (float*) pi;

- Integer와 같은 크기이기 때문에 포인터로 해석되지 않게끔 유의한다.

## malloc

Q1. malloc은 void\*를 반환하는데 int\* 나 float\* 로 명시적으로 캐스팅을 하는 이유?

- 현대 C에서는 void*가 암묵적으로 변환되기 때문에 명시적 캐스팅을 자제할 것을 권하기도 한다.

Q2. ANSI C 이전의 시스템에서 char*를 malloc의 결과로 사용했던 이유?

- char은 크기가 1byte이고 포인터 또한 연산이 되니까 메모리 순회하기 좋았다. 따라서 컨벤션으로 사용하기도 했다.

Q3. free 함수에서 casting을 생략하는 이유

- malloc이 호출될 때 메모리 할당 뿐만 아니라 크기 등의 메타데이터를 추가적인 리소스를 들여 저장한다. 이후에 free가 void*로 인자를 받더라도 이전의 메타데이터를 참고하기 때문에 잘 동작한다.

```cpp
struct Node {
    int data;
    Node* next;
};

Node* head = nullptr;

Node* createNode(int value) {
    return new Node{value, nullptr};
}

void push_front(Node*& head, int value) {
    head = new Node{value, head};
}

void push_back(Node*& head, int value) {
    if(head == nullptr) {
        Node* newNode = new Node{value, nullptr};
        head = newNode;
        return;
    }
    
    Node* cur = head;
    while(cur->next != nullptr) {
        cur = cur->next;
    }
    Node* newNode = createNode(value);
    cur->next = newNode;
}

void printList(Node* head) {
    Node* cur = head;
    while(cur != nullptr) {
        cout << cur->data << " ";
        cur = cur->next;
    }
}

bool remove(Node*& head, int value) {
    Node* cur = head;
    Node* prev = nullptr;
    while(cur != nullptr) {
        if(cur->data == value) {
            if(prev != nullptr) {
                prev->next = cur->next;
            } else {
                head = cur->next;
            }
            delete cur;
            return true;
        }
        prev = cur;
        cur = cur->next;
    }
    return false;
}

Node* find(Node* head, int value) {
    Node* cur = head;
    while(cur != nullptr) {
        if(cur->data == value) {
            return cur;
        }
        cur = cur->next;
    }
    return nullptr;
}

bool insert_at(Node*& head, int index, int value) {
    if (index < 0) return false;

    if (index == 0) {
        head = new Node{value, head};
        return true;

    Node* cur = head;
    int current_idx = 0;

    while (cur != nullptr && current_idx < index - 1) {
        cur = cur->next;
        current_idx++;

    return true;
```

```cpp
struct Node {
    int coef;
    int expon;
    Node* next;

Node* pread(int coef[], int expon[], int n) {
    Node* phead = nullptr;
    for(int i = n - 1; i >= 0; i--) {
        phead = new Node{coef[i], expon[i], phead};
    }
    return phead;
}

void pwrite(Node* head) {
    Node* cur = head;
    while(cur != nullptr) {
        cout << ((cur->coef) > 0 ? ((cur == head) ? "" : " + ") : " - ")
        << abs(cur->coef)
        << ((cur->expon > 0) ? "x^" : "")
        << ((cur->expon > 0) ? to_string(cur->expon) : "");
        cur = cur->next;
    }
}

Node* padd(Node* a_head, Node* b_head) {
    Node* a_cur = a_head;
    Node* b_cur = b_head;
    Node dummy{0, 0, nullptr};
    Node* res_cur = &dummy;

    int res_coef, res_expon;
    while(a_cur != nullptr || b_cur != nullptr) {
        if(a_cur != nullptr && (b_cur == nullptr || a_cur->expon > b_cur->expon)) {
            res_coef = a_cur->coef;
            res_expon = a_cur->expon;
            a_cur = a_cur->next;
        }
        else if(b_cur != nullptr && (a_cur == nullptr || a_cur->expon < b_cur->expon)) {
            res_coef = b_cur->coef;
            res_expon = b_cur->expon;
            b_cur = b_cur->next;
        }
        else {
            res_coef = a_cur->coef + b_cur->coef;
            res_expon = a_cur->expon;
            a_cur = a_cur->next;
            b_cur = b_cur->next;
            if(res_coef == 0) continue;
        }

        res_cur->next = new Node{res_coef, res_expon, nullptr};
        res_cur = res_cur->next;
    }
    return dummy.next;
}

Node* psub(Node* a_head, Node* b_head) {
    Node* a_cur = a_head;
    Node* b_cur = b_head;
    Node dummy{0, 0, nullptr};
    Node* res_cur = &dummy;

    int res_coef, res_expon;
    while(a_cur != nullptr || b_cur != nullptr) {
        if(a_cur != nullptr && (b_cur == nullptr || a_cur->expon > b_cur->expon)) {
            res_coef = a_cur->coef;
            res_expon = a_cur->expon;
            a_cur = a_cur->next;
        }
        else if(b_cur != nullptr && (a_cur == nullptr || a_cur->expon < b_cur->expon)) {
            res_coef = -b_cur->coef;
            res_expon = b_cur->expon;
            b_cur = b_cur->next;
        }
        else {
            res_coef = a_cur->coef - b_cur->coef;
            res_expon = a_cur->expon;
            a_cur = a_cur->next;
            b_cur = b_cur->next;
            if(res_coef == 0) continue;

        res_cur->next = new Node{res_coef, res_expon, nullptr};
        res_cur = res_cur->next;
    }
    return dummy.next;
}

Node* pmul(Node* a_head, Node* b_head) {
    if(a_head == nullptr || b_head == nullptr) return nullptr;
    int max_expon = a_head->expon + b_head->expon;
    vector<int> res_coef(max_expon + 1, 0);
    Node* a_cur = a_head;
    Node* b_cur = b_head;
    while(a_cur != nullptr) {
        while(b_cur != nullptr) {
            int cur_expon = a_cur->expon + b_cur->expon; 
            int cur_coef = a_cur->coef * b_cur->coef;
            res_coef[cur_expon] += cur_coef;
            b_cur = b_cur->next;
        }
        b_cur = b_head;
        a_cur = a_cur->next;
    }

    Node dummy{0, 0, nullptr};
    Node* res_cur = &dummy;
    for(int i = max_expon; i >= 0; i--) {
        if(res_coef[i] != 0) {
            res_cur->next = new Node{res_coef[i], i, nullptr};
            res_cur = res_cur->next;
        }
    }
    return dummy.next;
}

float eval(Node* head, float n) {
    Node* cur = head;
    float res = 0;
    while(cur != nullptr) {
        res += cur->coef * pow(n, cur->expon);
        cur = cur->next;
    }
    return res;
}

void perase(Node*& head) {
    while(head != nullptr) {
        Node* temp = head;
        head = temp->next;
        delete temp;
    }
}
```

- 만약에 곱셈을 벡터 안쓰고 linked list만으로 구현한다면... $O(n^3)$ : 곱셈으로 $n^2$ term 계산 $O(n^2)$ + 각 term에 대해 링크드 리스트를 순회하면서 insert할지 더해줄지 판단 $O(n)$

```cpp
struct Node {
    int coef;
    int expon;
    Node* next;
};


Node* pread(int coef[], int expon[], int n) {
    Node* head = new Node{0, -1, nullptr};
    head->next = head;
    Node* cur = head;
    for(int i = 0; i < n; i++) {
        cur->next = new Node{coef[i], expon[i], nullptr};
        cur = cur->next;
    }
    cur->next = head;
    return head;
}

void pwrite(Node* head) {
    if(head->next == head) return;
    Node* cur = head->next;

        cout << ((cur->coef) > 0 ? ((cur == head->next) ? "" : " + ") : " - ")

    Node* a_cur = a_head->next;
    Node* b_cur = b_head->next;
    Node* res_head = new Node{0, -1, nullptr};
    Node* res_cur = res_head;

    int res_coef, res_expon;
    while(a_cur != a_head || b_cur != b_head) {
        if(a_cur != a_head && (b_cur == b_head || a_cur->expon > b_cur->expon)) {

            res_expon = a_cur->expon;

            res_coef = b_cur->coef;
            res_expon = b_cur->expon;
            b_cur = b_cur->next;

            res_expon = a_cur->expon;

            b_cur = b_cur->next;
            if(res_coef == 0) continue;

        res_cur->next = new Node{res_coef, res_expon, nullptr};

            res_coef[cur_expon] += cur_coef;
            b_cur = b_cur->next;

        b_cur = b_head->next;
        a_cur = a_cur->next;

    Node* res_head = new Node{0, -1, nullptr};
    Node* res_cur = res_head;
    for(int i = max_expon; i >= 0; i--) {
        if(res_coef[i] != 0) {
            res_cur->next = new Node{res_coef[i], i, nullptr};

        cur = cur->next;
    }
    return res;
}

void perase(Node*& head) {
    Node* head_cp = head;
    head = head->next;
    while(head != head_cp) {
        Node* temp = head;
        head = temp->next;
        delete temp;
    }
    delete head_cp;
    head = nullptr;
}
```

```cpp
Node* invert(Node* head) {
    Node* cur = head;
    Node* middle;
    Node* trail = nullptr;
    while(cur != nullptr) {
        middle = cur;
        cur = cur->next;
        middle->next = trail;
        trail = middle;
```

```cpp
Node* concat(Node*& a_head, Node*& b_head) {
    if(a_head == nullptr) return b_head;
    else {
        Node* a_cur = a_head;
        while(a_cur->next != nullptr) {
            a_cur = a_cur->next;
        }
        a_cur->next = b_head;
    }
    return a_head;
}
```

# 4.6 Equivalence Relations
동치(Equivalence)는 다음 세 조건을 만족한다.
1. x = x
2. x = y, then y = x
3. x = y and y = z, then x = z

### 갑자기 이 개념이 왜 나왔는가
n개의 pair <i, j>에 대해 동치 관계를 2차원 배열에 저장하려고 하면 n^2의 공간이 필요하다. 메모리 낭비가 되기 때문에 대신 dim-n의 배열에 각 linked list의 head가 오도록 저장하여 메모리를 아낄 수 있다. (seq[i] insert j, seq[j] insert i)

이후 DFS, BFS에서 쓰이는 그래프의 연결 요소 출력과도 연관된다.

# 4.7 Sparse Matrices

```cpp
struct Node {
    enum class Tag {HEAD_NODE, ENTRY_NODE} tag;
    Node* right;
    Node* down;

    union {
        struct Entry {
            int row;
            int col;
            int value;
        } entry;
        struct Head {
            struct Node* next;
        } head;
    };
};

Node* make_head() {
    Node* newNode = new Node{};
    newNode->tag = Node::Tag::HEAD_NODE;
    newNode->right = nullptr;
    newNode->down = nullptr;
    newNode->head.next = nullptr;
    return newNode;
}

Node* make_entry(int row, int col, int value) {
    Node* newNode = new Node{};
    newNode->tag = Node::Tag::HEAD_NODE;
    newNode->right = nullptr;
    newNode->down = nullptr;
    newNode->entry = {row, col, value};
    return newNode;
}

void mread() {
    int n_row, n_col, n_data;
    cin >> n_row >> n_col >> n_data;

    int n_head = max(n_row, n_col);
    Node* head_node[n_head];
    Node* m_pointer = make_head();
    for(int i = 0; i < n_head; i++) {
        head_node[i] = make_head();
        head_node[i]->right = head_node[i];
        head_node[i]->down = head_node[i];
        if(i == n_head - 1) head_node[i]->head.next = m_pointer;
        else head_node[i]->head.next = head_node[i + 1];
    }
    m_pointer->head.next = head_node[0];

    for(int i = 0; i < n_data; i++) {
        int row, col, value;
        cin >> row >> col >> value;
        Node* newEntry = make_entry(row, col, value);
        Node* cur = head_node[row];
        while(cur->right != head_node[row] && cur->right->entry.col < newEntry->entry.col) {
            cur = cur->right;
        }
        newEntry->right = cur->right;
        cur->right = newEntry;
        
        cur = head_node[col];
        while(cur->down != head_node[col] && cur->down->entry.row < newEntry->entry.row) {
            cur = cur->down;
        }
        newEntry->down = cur->down;
        cur->down = newEntry;
    }
}
```

# 4.8 Doubly Linked Lists
### Circular
```cpp
struct Node {
    int value;
    Node* llink;
    Node* rlink;
};

Node* create_list() {
    Node* head = new Node{};
    head->llink = head;
    head->rlink = head;
    return head;
}

void delete_node(Node*& node) {
    Node* left = node->llink;
    Node* right = node->rlink;
    left->rlink = right;
    right->llink = left;
    delete node;
}

void insert_after(Node*& node, int value) {
    Node* newNode = new Node{value, node, node->rlink}
    node->rlink->llink = newNode;
    node->rlink = newNode;
}
```

### 어떤 장단점이 있을까
- 맨 앞 뒤 insert에 둘 다 O(1)
- 특정 노드만 주어져도 삭제가 가능하다(굳이 이전 노드를 기억할 필요가 없다)
- 메모리 포인터 하나 더 필요하다
