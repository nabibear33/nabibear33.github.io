## 문제 한줄요약

왕복 문제를 해결하기 위해 다익스트라를 두번 돌려서 해결할 수 있다

그런데 처음 풀었던 풀이는 의도치 않게 spfa 알고리즘이 되어 queue만으로 구현했는데 풀리게 되었다.


## 문제
세준이는 등산광이다. 세준이는 높은 곳에서 도시를 내려다 보는 것을 좋아한다. 하지만, 겁이 많은 세준이는 어두워지기 전에 호텔로 내려오려고 한다.

세준이가 가려고하는 산의 지도가 입력으로 주어진다. 산의 지도를 M이라고 했을 때, M[i][j]는 (i,j)의 높이가 M[i][j]라는 것을 의미한다. 그 값이 'A'-'Z'일 때는 0-25를 뜻하는 것이고, 'a'-'z'일 때는, 26-51을 뜻한다.

세준이의 호텔은 (0,0)에 있다. 그리고, 세준이는 지금 위치에서 바로 인접한 정수 좌표 중 높이의 차이가 T보다 크지 않은 곳으로만 다닐 수 있다.

만약 세준이가 현재 위치에서 높이가 낮은 곳이나 같은 곳으로 이동한다면 시간은 1초가 걸린다. 하지만 높은 곳으로 이동한다면 두 위치의 높이의 차이의 제곱만큼 시간이 걸린다. 예를 들어 높이가 5에서 높이가 9인 곳으로 간다면, 시간은 (5-9)2=16초가 걸린다. 하지만, 높이가 9인 곳에서 5인 곳으로 간다면 시간은 1초가 걸린다.

산의 지도와, T, 그리고 어두워지는 시간 D가 주어졌을 때, 세준이가 D보다 크지 않은 시간 동안 올라갈 수 있는 최대 높이를 구하는 프로그램을 작성하시오.(세준이는 호텔에서 출발해서 호텔로 돌아와야 한다.)

첫째 줄에 산의 세로크기 N과 가로크기 M 그리고, T와 D가 주어진다. N과 M은 25보다 작거나 같은 자연수이다. 둘째 줄부터 N개의 줄에 지도가 주어진다. T는 52보다 작거나 같은 자연수이고, D는 1,000,000보다 작거나 같은 자연수이다.

첫째 줄에 세준이가 갈 수 있는 가장 높은 곳의 높이를 출력한다.




## 아이디어
처음에는 BFS에 여러 조건을 걸면 풀릴 것 같았다. 틀리길래 나중에 우선순위 큐를 사용하는 다익스트라로 코드를 다시 짰다. 하지만 문제는 그냥 위에 depth 계산하는 함수를 잘못 짰던 것... 사실 둘 다 통과되는 코드였다.

## 왜 SPFA 알고리즘을 만족했는지?
우선 최솟값을 저장하는 time[][] 배열이 존재한다. 최솟값보다 작을 때에만 q에 넣기 때문에 최단거리 보장을 깨지 않게 된다. 

```
function SPFA(Graph, source):
    for each vertex v in Graph:
        dist[v] = INF
        inQueue[v] = false

    dist[source] = 0
    create empty queue Q
    Q.push(source)
    inQueue[source] = true

    while Q is not empty:
        u = Q.pop()
        inQueue[u] = false

        for each edge (u → v) with weight w:
            if dist[u] + w < dist[v]: \\ 요기
                dist[v] = dist[u] + w

                if inQueue[v] == false:
                    Q.push(v)
                    inQueue[v] = true

    return dist
```

다익스트라와 같이 relaxation이 일어나지만 거리순으로 정렬하여 한번 방문한 노드는 절대 다루지 않는 우선순위 큐와 다르게 이건 relaxation 조건을 조금 유하게 두어 음수 간선에도 대응할 수 있도록 한 것.

```cpp
#include <bits/stdc++.h>
#define DEBUG
using namespace std;


struct State {
    int r;
    int c;
    int d;
};


struct Delta {
    int r;
    int c;
};


class Board {
    private:
        vector<vector<int>> v;
        vector<vector<int>> asc_time;
        vector<vector<int>> dec_time;
        int n, m, t, d_limit;
        vector<Delta> deltas = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
        bool valid(int r, int c);
        int depth(State &current, State &next, string &s);
        
    public:
        Board(vector<vector<int>> &v_, int t_, int d_);
        void simulate();

};


Board::Board(vector<vector<int>> &v_, int t_, int d_) : v(v_), t(t_), d_limit(d_) {
    n = v.size();
    m = v[0].size();
    asc_time = vector<vector<int>>(n, vector<int>(m, -1));
    asc_time[0][0] = 0;
    dec_time = vector<vector<int>>(n, vector<int>(m, -1));
    dec_time[0][0] = 0;
}


bool Board::valid(int r, int c) {
    return (r >= 0) && (r < n) && (c >= 0) && (c < m);
}


int Board::depth(State &current, State &next, string &s) {
    if(s == "ASC") {
        if(v[current.r][current.c] >= v[next.r][next.c]) {
            return 1;
        } else {
            return (v[current.r][current.c] - v[next.r][next.c]) * (v[current.r][current.c] - v[next.r][next.c]);
        }
    }
    if(s == "DEC") {
        if(v[current.r][current.c] <= v[next.r][next.c]) {
            return 1;
        } else {
            return (v[current.r][current.c] - v[next.r][next.c]) * (v[current.r][current.c] - v[next.r][next.c]);
        }
    }
}


void Board::simulate() {
    queue<State> q;
    q.push(State{0, 0});
    string mode = "ASC";

    while(!q.empty()) {
        State current = q.front();
        q.pop();

        for(Delta &d : deltas) {
            if(valid(current.r + d.r, current.c + d.c)) {
                State next = {current.r + d.r, current.c + d.c, current.d};
                if(abs(v[current.r][current.c] - v[next.r][next.c]) <= t) {
                    next.d += depth(current, next, mode);
                    if((asc_time[next.r][next.c] == -1 || next.d < asc_time[next.r][next.c]) && next.d <= d_limit) {
                        asc_time[next.r][next.c] = next.d;
                        q.push(next);
                    }
                }
            }
        }
    }

    q.push(State{0, 0});
    mode = "DEC";

    while(!q.empty()) {
        State current = q.front();
        q.pop();

        for(Delta &d : deltas) {
            if(valid(current.r + d.r, current.c + d.c)) {
                State next = {current.r + d.r, current.c + d.c, current.d};
                if(abs(v[current.r][current.c] - v[next.r][next.c]) <= t) {
                    next.d += depth(current, next, mode);
                    if((dec_time[next.r][next.c] == -1 || next.d < dec_time[next.r][next.c]) && next.d <= d_limit) {
                        dec_time[next.r][next.c] = next.d;
                        q.push(next);
                    }
                }
            }
        }
    }

    int max_height = 0;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(asc_time[i][j] != -1 && dec_time[i][j] != -1 && asc_time[i][j] + dec_time[i][j] <= d_limit) {
                max_height = max(max_height, v[i][j]);
            }
        }
    }

    cout << max_height;

}


int main() {
    int n, m, t, d;
    cin >> n >> m >> t >> d;

    vector<vector<int>> v(n, vector<int>(m));
    for(int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for(int j = 0; j < m; j++) {
            if(s[j] - 'Z' <= 0) v[i][j] = s[j] - 'A';
            if(s[j] - 'a' >= 0) v[i][j] = s[j] - 'a' + 26;
        }
    }

    Board b(v, t, d);
    b.simulate();

    return 0;
}
```

## Priority queue Dijkstra
```cpp
#include <bits/stdc++.h>
#define DEBUG
using namespace std;


struct State {
    int r;
    int c;
    int d;
};


struct Compare {
    bool operator()(const State &a, const State &b) const {
        return a.d > b.d;
    }
};


struct Delta {
    int r;
    int c;
};


class Board {
    private:
        vector<vector<int>> v;
        vector<vector<int>> asc_time;
        vector<vector<int>> dec_time;
        int n, m, t, d_limit;
        vector<Delta> deltas = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
        bool valid(int r, int c);
        int depth(State &current, State &next, string &s);
        
    public:
        Board(vector<vector<int>> &v_, int t_, int d_);
        void simulate();

};


Board::Board(vector<vector<int>> &v_, int t_, int d_) : v(v_), t(t_), d_limit(d_) {
    n = v.size();
    m = v[0].size();
    asc_time = vector<vector<int>>(n, vector<int>(m, -1));
    asc_time[0][0] = 0;
    dec_time = vector<vector<int>>(n, vector<int>(m, -1));
    dec_time[0][0] = 0;
}


bool Board::valid(int r, int c) {
    return (r >= 0) && (r < n) && (c >= 0) && (c < m);
}


int Board::depth(State &current, State &next, string &s) {
    if(s == "ASC") {
        if(v[current.r][current.c] >= v[next.r][next.c]) {
            return 1;
        } else {
            return (v[current.r][current.c] - v[next.r][next.c]) * (v[current.r][current.c] - v[next.r][next.c]);
        }
    }
    if(s == "DEC") {
        if(v[current.r][current.c] <= v[next.r][next.c]) {
            return 1;
        } else {
            return (v[current.r][current.c] - v[next.r][next.c]) * (v[current.r][current.c] - v[next.r][next.c]);
        }
    }
    
}


void Board::simulate() {
    priority_queue<State, vector<State>, Compare> pq;
    pq.push(State{0, 0});
    string mode = "ASC";

    while(!pq.empty()) {
        State current = pq.top();
        pq.pop();

        if(asc_time[current.r][current.c] < current.d) {
            continue;
        }

        for(Delta &d : deltas) {
            if(valid(current.r + d.r, current.c + d.c)) {
                State next = {current.r + d.r, current.c + d.c, current.d};
                if(abs(v[current.r][current.c] - v[next.r][next.c]) <= t) {
                    next.d += depth(current, next, mode);
                    if((asc_time[next.r][next.c] == -1 || next.d < asc_time[next.r][next.c]) && next.d <= d_limit) {
                        asc_time[next.r][next.c] = next.d;
                        pq.push(next);
                    }
                }
            }
        }
    }

    pq.push(State{0, 0});
    mode = "DEC";

    while(!pq.empty()) {
        State current = pq.top();
        pq.pop();

        if(dec_time[current.r][current.c] < current.d) {
            continue;
        }

        for(Delta &d : deltas) {
            if(valid(current.r + d.r, current.c + d.c)) {
                State next = {current.r + d.r, current.c + d.c, current.d};
                if(abs(v[current.r][current.c] - v[next.r][next.c]) <= t) {
                    next.d += depth(current, next, mode);
                    if((dec_time[next.r][next.c] == -1 || next.d < dec_time[next.r][next.c]) && next.d <= d_limit) {
                        dec_time[next.r][next.c] = next.d;
                        pq.push(next);
                    }
                }
            }
        }
    }

    int max_height = 0;
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < m; j++) {
            if(asc_time[i][j] != -1 && dec_time[i][j] != -1 && asc_time[i][j] + dec_time[i][j] <= d_limit) {
                max_height = max(max_height, v[i][j]);
            }
        }
    }

    cout << max_height;

}


int main() {
    int n, m, t, d;
    cin >> n >> m >> t >> d;

    vector<vector<int>> v(n, vector<int>(m));
    for(int i = 0; i < n; i++) {
        string s;
        cin >> s;
        for(int j = 0; j < m; j++) {
            if(s[j] - 'Z' <= 0) v[i][j] = s[j] - 'A';
            if(s[j] - 'a' >= 0) v[i][j] = s[j] - 'a' + 26;
        }
    }

    Board b(v, t, d);
    b.simulate();

    return 0;
}
```
