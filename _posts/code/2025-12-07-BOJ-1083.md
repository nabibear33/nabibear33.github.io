## 문제
많은 공장에서 로봇이 이용되고 있다. 우리 월드 공장의 로봇은 바라보는 방향으로 궤도를 따라 움직이며, 움직이는 방향은 동, 서, 남, 북 가운데 하나이다. 로봇의 이동을 제어하는 명령어는 다음과 같이 두 가지이다.

명령 1. Go k: k는 1, 2 또는 3일 수 있다. 현재 향하고 있는 방향으로 k칸 만큼 움직인다.
명령 2. Turn dir: dir은 left 또는 right 이며, 각각 왼쪽 또는 오른쪽으로 90° 회전한다.
공장 내 궤도가 설치되어 있는 상태가 아래와 같이 0과 1로 이루어진 직사각형 모양으로 로봇에게 입력된다. 0은 궤도가 깔려 있어 로봇이 갈 수 있는 지점이고, 1은 궤도가 없어 로봇이 갈 수 없는 지점이다. 로봇이 (4, 2) 지점에서 남쪽을 향하고 있을 때,  이 로봇을 (2, 4) 지점에서 동쪽으로 향하도록 이동시키는 것은 아래와 같이 9번의 명령으로 가능하다.

로봇의 현재 위치와 바라보는 방향이 주어졌을 때, 로봇을 원하는 위치로 이동시키고, 원하는 방향으로 바라보도록 하는데 최소 몇 번의 명령이 필요한지 구하는 프로그램을 작성하시오.

## 아이디어
회전하고 직진하는 두 명령을 처음에는 simulation 함수에서 함께 처리하려다 보니 예외적인 경우도 생기고 그런 부분에서 시간을 많이 잡아먹었다. 그러다가 나는 명령 두 개를 함수로 나누었다. 여전히 회전-직진의 set로 queue를 처리하도록 했지만, 직진 함수에서만 최솟값 비교(pruning) 및 다음 상태를 queue에 넘겨주는 작업을 진행하였다. 직진 함수에 직진하지 않는(i=0) 부분도 포함한 것이 최솟값 테이블을 더 안정적으로 구성하고 커버리지를 넓히는데 도움이 많이 되었다.

아울러 디버깅을 위해 구조체 출력 함수 오버로드도 잘 짜두면 좋을 것 같다.


```cpp
//  start at first position
//  q <- start state
//  while q is not empty
//      rotate and go straight
//      save only valid state with minimum cost
```

## Code
```cpp
#include <bits/stdc++.h>
// #define DEBUG 
using namespace std;


struct State {
    int r;
    int c;
    int dir;
    int count;

    bool operator == (const State &o) const {
        return (r == o.r) && (c == o.c) && (dir == o.dir) && (count == o.count);
    }
};


ostream& operator<<(ostream& os, const State& s) {
    os << "State{ r=" << s.r
        << ", c=" << s.c
        << ", dir=" << s.dir
        << ", count=" << s.count
        << " }";
return os;
}


struct Delta {
    int dr;
    int dc;
};

class Board {
    private:
        vector<vector<int>> v;
        vector<vector<vector<int>>> min_table;  // min distance. third idx is dir
        int R, C;
        State start;
        State goal;
        map<int, Delta> deltas = {{1, {0, 1}}, {2, {0, -1}}, {3, {1, 0}}, {4, {-1, 0}}};
        int direction_offset(State &current, int dir);
        bool state_is_valid(State &current, Delta &d, int i);
        void go(queue<State> &s, State &current);
        void turn(queue<State> &s, State &current);

    public:
        Board(vector<vector<int>> &v, State &start_, State &goal_);
        ~Board() {};
        void simulate();
};


Board::Board (vector<vector<int>> &v_, State &start_, State &goal_) : v(v_), start(start_), goal(goal_) {
    R = v.size();
    C = v[0].size();
    min_table = vector<vector<vector<int>>>(R, vector<vector<int>>(C, vector<int>(5, INT_MAX)));
    min_table[start.r][start.c][start.dir] = 0;
}

int Board::direction_offset(State &current, int dir) {
    switch(dir) {
        case 1:
            if(current.dir == 1) return 0;
            if(current.dir == 2) return 2;
            else return 1;
            break;
        case 2:
            if(current.dir == 2) return 0;
            if(current.dir == 1) return 2;
            else return 1;
            break;
        case 3:
            if(current.dir == 3) return 0;
            if(current.dir == 4) return 2;
            else return 1;
            break;
        case 4:
            if(current.dir == 4) return 0;
            if(current.dir == 3) return 2;
            else return 1;
            break;
        default:
            throw runtime_error("Invalid direction offset\n");
    }
}


bool Board::state_is_valid(State &current, Delta &d, int i) {
    if(current.r + i * d.dr >= 0 && current.r + i * d.dr < R && current.c + i * d.dc >= 0 && current.c + i * d.dc < C) {
        return (v[current.r + i * d.dr][current.c + i * d.dc] == 0) ? true : false;
    } else {
        return false;
    }
}


void Board::go(queue<State> &s, State &current) {
    #ifdef DEBUG
    cerr << "start go\n";
    #endif
    for(int i = 0; i <= 3; i++) {
        if(state_is_valid(current, deltas[current.dir], i)) {
            State next = {current.r + i * deltas[current.dir].dr, current.c + i * deltas[current.dir].dc, current.dir, current.count + ((i != 0) ? 1 : 0)};
            #ifdef DEBUG
            cerr << next << "\n";
            #endif
            if(min_table[next.r][next.c][next.dir] > next.count) {
                #ifdef DEBUG
                cerr << "update table... " << "value comparison : " << next.count << " vs " << min_table[next.r][next.c][next.dir] << "\n";
                #endif
                min_table[next.r][next.c][next.dir] = next.count;
                if(i != 0) {
                    s.push(next);
                }
            }
        } else {
            break;
            #ifdef DEBUG
            cerr << "NOT VALID: Pass next state : (" << current.r + i * deltas[current.dir].dr << ", " << current.c + i * deltas[current.dir].dc << "), dir : " << current.dir << "\n";
            #endif
        }
    }
    #ifdef DEBUG
    cerr << "end go\n";
    #endif
}


void Board::turn(queue<State> &s, State &current) {
    #ifdef DEBUG
    cerr << "start turn\n";
    #endif
    for(auto &[dir, delta] : deltas) {
        State next = {current.r, current.c, dir, current.count + direction_offset(current, dir)};
        #ifdef DEBUG
        cerr << next << "\n";
        #endif
        go(s, next);
    }
    #ifdef DEBUG
    cerr << "end turn\n";
    #endif
}


void Board::simulate() {
    queue<State> q;
    q.push(start);

    while(!q.empty()) {
        State current = q.front();
        q.pop();
        #ifdef DEBUG
        cerr << "current " << current << "\n";
        #endif

        turn(q, current);
    }

    cout << min_table[goal.r][goal.c][goal.dir];
}


int main() {
    int M, N;
    cin >> M >> N;

    vector<vector<int>> v(M, vector<int>(N));
    for(int i = 0; i < M; i++) {
        for(int j = 0; j < N; j++) {
            cin >> v[i][j];
        }
    }
    
    State start;
    cin >> start.r >> start.c >> start.dir;
    start.r -= 1; start.c -= 1; start.count = 0;
    State goal;
    cin >> goal.r >> goal.c >> goal.dir;
    goal.r -= 1; goal.c -= 1; goal.count = 0;

    Board board(v, start, goal);
    board.simulate();

    return 0;
}
```
