## 문제
여행을 떠난 세준이는 지도를 하나 구하였다. 이 지도는 아래 그림과 같이 직사각형 모양이며 여러 칸으로 나뉘어져 있다. 한 칸은 한 지점을 나타내는데 각 칸에는 그 지점의 높이가 쓰여 있으며, 각 지점 사이의 이동은 지도에서 상하좌우 이웃한 곳끼리만 가능하다.

현재 제일 왼쪽 위 칸이 나타내는 지점에 있는 세준이는 제일 오른쪽 아래 칸이 나타내는 지점으로 가려고 한다. 그런데 가능한 힘을 적게 들이고 싶어 항상 높이가 더 낮은 지점으로만 이동하여 목표 지점까지 가고자 한다. 위와 같은 지도에서는 다음과 같은 세 가지 경로가 가능하다.

지도가 주어질 때 이와 같이 제일 왼쪽 위 지점에서 출발하여 제일 오른쪽 아래 지점까지 항상 내리막길로만 이동하는 경로의 개수를 구하는 프로그램을 작성하시오.

첫째 줄에는 지도의 세로의 크기 M과 가로의 크기 N이 빈칸을 사이에 두고 주어진다. 이어 다음 M개 줄에 걸쳐 한 줄에 N개씩 위에서부터 차례로 각 지점의 높이가 빈 칸을 사이에 두고 주어진다. M과 N은 각각 500이하의 자연수이고, 각 지점의 높이는 10000이하의 자연수이다.

첫째 줄에 이동 가능한 경로의 수 H를 출력한다. 모든 입력에 대하여 H는 10억 이하의 음이 아닌 정수이다.

## 아이디어
처음에는 bottom up으로 도착지점까지 가는 횟수의 총합을 구하려고 했다. 그러면 인접한 오르막인 길 찾기, 현재 칸의 dp를 채우기 위해 인접한 내리막인 길 찾기, 내리막인 길이 모두 계산되어 있으면 현재 칸의 dp 채우기, 위 과정을 반복... 인데 그러려면 막다른 길에 대한 체크 등 의존성에 대해 추가적으로 작업을 해 주어야 한다. 중복으로 들어가는 칸을 막기 위해 visit 배열을 사용하였다. 그리고 속도에 대한 최적화를 위해 높이가 낮은 칸을 더 빨리 큐에서 빼도록 우선순위 큐를 사용하였다.

반면 Top down으로 구성하게 되는 경우 의존성 문제가 자동으로 해결된다는 장점이 있다. (dfs로 재귀호출 때려버리면 막다른 길에서는 알아서 멈추게 됨).

visit 배열을 쓰는 대신 dp table에서 초기화를 -1로 하여 미계산 상태로 처리할 수 있다. 메모이제이션 하는 DFS에서 자주 쓴다고 한다. 0은 실제 계산 결과 경로가 없을 때.

## Bottom up BFS

```cpp
#include <bits/stdc++.h>
#define DEBUG
using namespace std;


struct Pos {
    int r;
    int c;
};


struct Delta {
    int r;
    int c;
};


class Board {
    private:
        vector<vector<int>> v;
        vector<vector<int>> dp;
        vector<vector<int>> visited;
        int M, N;
        vector<Delta> deltas = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
        bool is_valid_idx(Pos &p, Delta &d, const string& s);
        bool try_calculate_dp(Pos &p);
        void fill_valid_state_zero();
        struct Compare {
            Board* board;
            Compare(Board* b) : board(b) {}
            bool operator()(const Pos& a, const Pos& b) const {
                return board->v[a.r][a.c] > board->v[b.r][b.c];
            }
        };
    
    public:
        Board(vector<vector<int>> &v_);
        void simulate();
};


Board::Board(vector<vector<int>> &v_) : v(v_) {
    M = v.size(); N = v[0].size();
    dp = vector<vector<int>>(M, vector<int>(N, -1));
    dp[M - 1][N - 1] = 1;
    fill_valid_state_zero();
    visited = vector<vector<int>>(M, vector<int>(N, 0));
}


bool Board::is_valid_idx(Pos &p, Delta &d, const string &s) {
    if((p.r + d.r >= 0) && (p.r + d.r < M) && (p.c + d.c >= 0) && (p.c + d.c < N)) {
        if(s == "INC") {
                return (v[p.r][p.c] < v[p.r + d.r][p.c + d.c]);
        }
        else if(s == "DEC") {
                return (v[p.r][p.c] > v[p.r + d.r][p.c + d.c]);
        }
    }
    return false;
}


void Board::fill_valid_state_zero() {
    queue<Pos> q;
    q.push(Pos{M - 1, N - 1});

    while(!q.empty()) {
        Pos current = q.front();
        q.pop();

        for(Delta &d: deltas) {
            if(!is_valid_idx(current, d, "INC")) {
                continue;
            }
            Pos next = Pos{current.r + d.r, current.c + d.c};
            if(dp[next.r][next.c] != -1) {
                continue;
            }
            dp[next.r][next.c] = 0;
            q.push(next);
        }
    }
}


bool Board::try_calculate_dp(Pos &p) {
    int sum = 0;
    for(Delta &d: deltas) {
        if(!is_valid_idx(p, d, "DEC") || dp[p.r + d.r][p.c + d.c] == -1) {
            continue;
        }
        else if(dp[p.r + d.r][p.c + d.c] == 0) {
            return false;
        }
        else { 
            sum += dp[p.r + d.r][p.c + d.c];
        }
    }
    dp[p.r][p.c] = sum;
    return true;
}


void Board::simulate() {
    priority_queue<Pos, vector<Pos>, Compare> pq{Compare(this)};
    pq.push(Pos{M - 1, N - 1});

    while(!pq.empty()) {
        Pos current = pq.top();
        pq.pop();

        if(dp[current.r][current.c] == 0) {
            bool flag = try_calculate_dp(current);
            if(!flag) {
                pq.push(current);
                continue;
            }
        }

        for(Delta &d: deltas) {
            if(!is_valid_idx(current, d, "INC")) {
                continue;
            }
            Pos next = Pos{current.r + d.r, current.c + d.c};
            if(!visited[next.r][next.c]) {
                visited[next.r][next.c] = 1;
                pq.push(next);
            }
        }
    }

    cout << ((dp[0][0] == -1) ? 0 : dp[0][0]);
}


int main() {
    int m, n;
    cin >> m >> n;

    vector<vector<int>> v(m, vector<int>(n));
    for(int i = 0; i < m; i++) {
        for(int j = 0; j < n; j++) {
            cin >> v[i][j];
        }
    }

    Board board(v);
    board.simulate();
    
    return 0;
}
```

## Top Down DFS (Recursion)
```cpp
#include <bits/stdc++.h>
#define DEBUG
using namespace std;


struct Pos {
    int r;
    int c;
};


struct Delta {
    int r;
    int c;
};


class Board {
    private:
        vector<vector<int>> v;
        vector<vector<int>> dp;
        int M, N;
        vector<Delta> deltas = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
        bool is_valid_idx(Pos &p, Delta &d);
        int dfs(Pos &p);
    
    public:
        Board(vector<vector<int>> &v_);
        void simulate();
};


Board::Board(vector<vector<int>> &v_) : v(v_) {
    M = v.size(); N = v[0].size();
    dp = vector<vector<int>>(M, vector<int>(N, -1));
}


bool Board::is_valid_idx(Pos &p, Delta &d) {
    if((p.r + d.r < 0) || (p.r + d.r >= M) || (p.c + d.c < 0) || (p.c + d.c >= N)) {
        return false;
    }
    return (v[p.r][p.c] > v[p.r + d.r][p.c + d.c]);
}


int Board::dfs(Pos &p) {
    if(p.r == M - 1 && p.c == N - 1) {  // return goal point
        return 1;
    }

    if(dp[p.r][p.c] != -1) return dp[p.r][p.c]; // return the value if it is already calculated

    dp[p.r][p.c] = 0;

    for(Delta &d : deltas) {
        if(is_valid_idx(p, d)) {
            Pos next{p.r + d.r, p.c + d.c};
            dp[p.r][p.c] += dfs(next);
        }
    }
    return dp[p.r][p.c];
}


void Board::simulate() {
    Pos start{0, 0};
    dfs(start);
    cout << dp[0][0];
}


int main() {
    int m, n;
    cin >> m >> n;

    vector<vector<int>> v(m, vector<int>(n));
    for(int i = 0; i < m; i++) {
        for(int j = 0; j < n; j++) {
            cin >> v[i][j];
        }
    }

    Board board(v);
    board.simulate();
    
    return 0;
}
```
